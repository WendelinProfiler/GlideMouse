cmake_minimum_required(VERSION 3.24)
project(GlideMouse VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "GlideMouse currently supports Windows x64 only")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  add_compile_options(/utf-8 /W3 /EHsc)
else()
  add_compile_options(-Wall -Wextra -Werror)
  add_link_options(-static)
endif()
add_executable(IslandMagicInput WIN32 src/IslandMagicInput.cpp)
target_link_libraries(IslandMagicInput PRIVATE user32 shell32 gdiplus bthprops cfgmgr32 dwmapi shcore advapi32)
add_executable(GlideMouseConnection WIN32 src/GlideMouseConnection.cpp)
target_link_libraries(GlideMouseConnection PRIVATE bthprops shell32 advapi32)
add_executable(GlideMouseDriverSetup src/GlideMouseDriverSetup.cpp)
target_link_libraries(GlideMouseDriverSetup PRIVATE setupapi)
add_executable(IslandMagicUsbBridge src/IslandMagicUsbBridge.cpp)
target_link_libraries(IslandMagicUsbBridge PRIVATE advapi32 ws2_32 ole32 bthprops cfgmgr32)
if(MINGW)
  target_link_options(IslandMagicInput PRIVATE -municode)
  target_link_options(IslandMagicUsbBridge PRIVATE -municode)
  target_link_options(GlideMouseConnection PRIVATE -municode)
  target_link_options(GlideMouseDriverSetup PRIVATE -municode)
endif()
enable_testing()
foreach(test Engine TapModes MouseUsbParser Timing)
  add_executable(Test${test} tests/Test${test}.cpp)
  add_test(NAME ${test} COMMAND Test${test})
endforeach()
# Assertions are intentional behavioural tests and must also run in Release.
if(MSVC)
  foreach(test Engine TapModes MouseUsbParser Timing)
    target_compile_options(Test${test} PRIVATE /UNDEBUG)
  endforeach()
else()
  foreach(test Engine TapModes MouseUsbParser Timing)
    target_compile_options(Test${test} PRIVATE -UNDEBUG)
  endforeach()
endif()
add_executable(TestPopup tests/TestPopup.cpp)
target_link_libraries(TestPopup PRIVATE gdiplus shell32 dwmapi shcore advapi32 cfgmgr32)
add_test(NAME Popup COMMAND TestPopup)
add_executable(TestWindowsSettings tests/TestWindowsSettings.cpp)
target_link_libraries(TestWindowsSettings PRIVATE advapi32 shell32 cfgmgr32)
add_test(NAME WindowsSettings COMMAND TestWindowsSettings)
add_test(NAME Installer COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File ${CMAKE_CURRENT_SOURCE_DIR}/tests/TestInstaller.ps1)
